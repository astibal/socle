cmake_minimum_required(VERSION 3.20)
project(socle_meter_tests LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(Threads REQUIRED)
add_executable(meter_tests ../../common/meter.cpp ../../common/tests/meter_tests.cpp)
target_compile_features(meter_tests PRIVATE cxx_std_17)
target_compile_options(meter_tests PRIVATE -Wall -Wextra -Werror)
target_include_directories(meter_tests PRIVATE ../..)
target_link_libraries(meter_tests PRIVATE GTest::gtest_main Threads::Threads)
include(GoogleTest)
gtest_discover_tests(meter_tests)
